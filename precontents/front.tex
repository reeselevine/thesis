% A predefined page
%%-----------------------------------------
%% Abstract page
%%-----------------------------------------
%
\makeatletter
\NewDocumentEnvironment{abstract}{}{
	\clearpage
	\pagestyle{plain}
	\phantomsection
	\addchaptertocentry{}{Abstract}
	\begin{center}
		{\textbf{\Large Abstract}\par\bigskip
		{\@title}\par\bigskip
		{\@author}}
	\end{center}
	\ifdefined\@keywords
		\paragraph*{Key words:} {\@keywords}
	\fi
	\par
}{
	\clearpage
}
\makeatother
%%-----------------------------------------
%% Dedication page
%%-----------------------------------------
%
\RenewDocumentEnvironment{dedication}{}{
	\clearpage
	\pagestyle{plain}
	\phantomsection
	\addchaptertocentry{}{Dedication}
	\begin{center}
		\ \ \ \
		\vfill
}{
		\vfill
	\end{center}
	\clearpage
}
%%-----------------------------------------
%% Acknowledgments page
%%-----------------------------------------
%
\NewDocumentEnvironment{acknowledgement}{}{
	\clearpage
	\pagestyle{plain}
	\phantomsection
	\addchaptertocentry{}{Acknowledgments}
	\begin{center}
		\textbf{\Large Acknowledgments}
	\end{center}
	\par
}{
	\clearpage
}
